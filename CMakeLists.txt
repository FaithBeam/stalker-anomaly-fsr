cmake_minimum_required(VERSION 3.20)
project(anomaly_fsr VERSION 1.0.0 LANGUAGES CXX)

# Builds bin/anomaly_fsr.dll for 64-bit STALKER Anomaly (DX11 renderer).
# Works with MSVC (Visual Studio 2019+) or MinGW-w64 (see cmake/mingw-w64-x86_64.cmake).

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "anomaly_fsr must be built as a 64-bit DLL (Anomaly is a 64-bit game).")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(FSR1_DIR ${CMAKE_CURRENT_SOURCE_DIR}/third_party/fsr1)
set(SHADER_DIR ${CMAKE_CURRENT_SOURCE_DIR}/src/shaders)

# The shader source and AMD's FSR headers are embedded into the DLL so the mod ships a single file.
# The HLSL is compiled at runtime with the system d3dcompiler_47.dll.
set(EMBED_HEADER ${CMAKE_CURRENT_BINARY_DIR}/generated/embedded_shaders.h)
set(EMBED_INPUTS
  "fsr.hlsl|${SHADER_DIR}/fsr.hlsl"
  "ffx_a.h|${FSR1_DIR}/ffx_a.h"
  "ffx_fsr1.h|${FSR1_DIR}/ffx_fsr1.h")

file(WRITE ${EMBED_HEADER} "// Generated by CMakeLists.txt - do not edit.\n#pragma once\n#include <cstddef>\n\n")
set(EMBED_TABLE "")
set(EMBED_INDEX 0)
foreach(entry IN LISTS EMBED_INPUTS)
  string(REPLACE "|" ";" parts "${entry}")
  list(GET parts 0 embed_name)
  list(GET parts 1 embed_path)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${embed_path})
  file(READ ${embed_path} embed_hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," embed_hex "${embed_hex}")
  file(APPEND ${EMBED_HEADER} "static const unsigned char kEmbedded${EMBED_INDEX}[] = {${embed_hex}0x00};\n")
  string(APPEND EMBED_TABLE "  {\"${embed_name}\", reinterpret_cast<const char*>(kEmbedded${EMBED_INDEX}), sizeof(kEmbedded${EMBED_INDEX}) - 1},\n")
  math(EXPR EMBED_INDEX "${EMBED_INDEX} + 1")
endforeach()
file(APPEND ${EMBED_HEADER} "\nstruct EmbeddedFile { const char* name; const char* data; size_t size; };\n")
file(APPEND ${EMBED_HEADER} "static const EmbeddedFile kEmbeddedFiles[] = {\n${EMBED_TABLE}};\n")

add_library(anomaly_fsr SHARED
  src/exports.cpp
  src/fsr_constants.cpp
  src/fsr_pass.cpp
  src/log.cpp
  src/swapchain_hooks.cpp)

target_include_directories(anomaly_fsr PRIVATE
  src
  ${FSR1_DIR}
  ${CMAKE_CURRENT_BINARY_DIR}/generated)

target_compile_definitions(anomaly_fsr PRIVATE
  WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE
  ANOMALY_FSR_VERSION="${PROJECT_VERSION}")

set_target_properties(anomaly_fsr PROPERTIES PREFIX "" OUTPUT_NAME "anomaly_fsr")

if(MSVC)
  target_compile_options(anomaly_fsr PRIVATE /W4 /permissive-)
  # Static CRT: the DLL must not depend on a particular VC++ redistributable being present.
  set_property(TARGET anomaly_fsr PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_link_libraries(anomaly_fsr PRIVATE dxguid)
else()
  # AMD's headers are compiled as-is; keep their warnings out of the build log.
  set_source_files_properties(src/fsr_constants.cpp PROPERTIES COMPILE_OPTIONS "-w")
  target_compile_options(anomaly_fsr PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers)
  # Fully static so the DLL only imports system libraries (no libstdc++/libgcc/winpthread DLLs).
  target_link_options(anomaly_fsr PRIVATE -static -static-libgcc -static-libstdc++ -s)
  target_link_libraries(anomaly_fsr PRIVATE dxguid uuid)
endif()

# End-to-end test executable (runs on Windows or under Wine): fsr_selftest.exe path\to\anomaly_fsr.dll
add_executable(fsr_selftest tests/fsr_selftest.cpp)
target_compile_definitions(fsr_selftest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(fsr_selftest PRIVATE d3d11 dxguid)
if(NOT MSVC)
  target_link_options(fsr_selftest PRIVATE -static -static-libgcc -static-libstdc++)
endif()
